`app/src/main/db.js`:

```js
import Database from 'better-sqlite3'

let db = null

export function initDb(dbPath) {
  db = new Database(dbPath)
  db.pragma('journal_mode = WAL')
  db.exec(`
    CREATE TABLE IF NOT EXISTS history (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      text TEXT NOT NULL,
      raw_text TEXT,
      app_name TEXT,
      duration_ms INTEGER,
      words INTEGER,
      ppm REAL,
      created_at TEXT DEFAULT (datetime('now', 'localtime'))
    );
    CREATE TABLE IF NOT EXISTS dictionary (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      term TEXT NOT NULL,
      replacement TEXT,
      created_at TEXT DEFAULT (datetime('now', 'localtime'))
    );
    CREATE TABLE IF NOT EXISTS snippets (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      trigger TEXT NOT NULL,
      content TEXT NOT NULL,
      created_at TEXT DEFAULT (datetime('now', 'localtime'))
    );
    CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT);
  `)
  return db
}

export function saveDictation({ text, rawText, appName, durationMs }) {
  const words = text.trim() ? text.trim().split(/\s+/).length : 0
  const ppm = durationMs > 0 ? Math.round((words / (durationMs / 60000)) * 10) / 10 : 0
  const info = db.prepare(
    'INSERT INTO history (text, raw_text, app_name, duration_ms, words, ppm) VALUES (?, ?, ?, ?, ?, ?)'
  ).run(text, rawText, appName, durationMs, words, ppm)
  return db.prepare('SELECT * FROM history WHERE id = ?').get(info.lastInsertRowid)
}

export function getLastDictation() {
  return db.prepare('SELECT * FROM history ORDER BY id DESC LIMIT 1').get()
}

export function listHistory(limit = 50) {
  return db.prepare('SELECT * FROM history ORDER BY id DESC LIMIT ?').all(limit)
}

export function getSetting(key, fallback = null) {
  const row = db.prepare('SELECT value FROM settings WHERE key = ?').get(key)
  return row ? row.value : fallback
}

export function setSetting(key, value) {
  db.prepare('INSERT INTO settings (key, value) VALUES (?, ?) ON CONFLICT(key) DO UPDATE SET value = excluded.value')
    .run(key, String(value))
}

export function listDictionary() {
  return db.prepare('SELECT * FROM dictionary ORDER BY id').all()
}
```

```js
const AUDIO_MARKERS = /\[(BLANK_AUDIO|MÚSICA|MUSIC|APPLAUSE|RUÍDO|NOISE|SILÊNCIO)\]/gi

function escapeRegExp(s) {
  return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
}

export function polish(rawText, { dictionary = [], snippets = [] } = {}) {
  let text = String(rawText || '').replace(AUDIO_MARKERS, ' ')
  text = text.replace(/\s+/g, ' ').trim()
  if (!text) return ''

  const normalized = text.toLowerCase().replace(/[.!?,;]+$/, '').trim()
  for (const s of snippets) {
    if (normalized === s.trigger.toLowerCase().trim()) return s.content
  }

  for (const d of dictionary) {
    if (!d.replacement) continue
    const re = new RegExp(`(^|[^\\p{L}\\p{N}])${escapeRegExp(d.term)}(?=$|[^\\p{L}\\p{N}])`, 'giu')
    text = text.replace(re, (m, pre) => pre + d.replacement)
  }

  text = text.replace(/\s+/g, ' ').trim()
  return text.charAt(0).toUpperCase() + text.slice(1)
}
```

- [ ] **Step 1: Escrever o helper**

`helper/DitaHotkeys.swift`:

```swift
import Cocoa
import CoreGraphics

// Estado compartilhado
final class State {
    static let shared = State()
    var recording = false            // setado via stdin, controla supressão do Esc
    var rightCmdDown = false
    var rightCmdCombo = false
}

func emit(_ event: String) {
    print("{\"event\":\"\(event)\"}")
    fflush(stdout)
}

// Thread de stdin: "state on" / "state off"
DispatchQueue.global().async {
    while let line = readLine(strippingNewline: true) {
        if line == "state on" { State.shared.recording = true }
        if line == "state off" { State.shared.recording = false }
    }
}

guard AXIsProcessTrusted() else {
    emit("no_permission")
    exit(1)
}

let kVK_Space: Int64 = 49
let kVK_Escape: Int64 = 53
let kVK_RightCommand: Int64 = 54

let mask: CGEventMask =
    (1 << CGEventType.keyDown.rawValue) | (1 << CGEventType.flagsChanged.rawValue)

let callback: CGEventTapCallBack = { _, type, event, _ in
    let keyCode = event.getIntegerValueField(.keyboardEventKeycode)
    let flags = event.flags
    let st = State.shared

    if type == .flagsChanged && keyCode == kVK_RightCommand {
        if flags.contains(.maskCommand) {
            st.rightCmdDown = true
            st.rightCmdCombo = false
            emit("ptt_down")
        } else if st.rightCmdDown {
            st.rightCmdDown = false
            emit(st.rightCmdCombo ? "ptt_abort" : "ptt_up")
        }
        return Unmanaged.passRetained(event)
    }

    if type == .keyDown {
        // Fn+Espaço: toggle. Suprime o evento pro sistema não digitar espaço.
        if keyCode == kVK_Space && flags.contains(.maskSecondaryFn) {
            emit("toggle")
            return nil
        }
        // Esc: só reporta (e suprime) se estiver gravando
        if keyCode == kVK_Escape && st.recording {
            emit("esc")
            return nil
        }
        // Qualquer tecla com right-cmd segurado = combo normal (Cmd+V etc.), aborta PTT
        if st.rightCmdDown {
            st.rightCmdCombo = true
            emit("combo")
        }
    }

    return Unmanaged.passRetained(event)
}

guard let tap = CGEvent.tapCreate(
    tap: .cgSessionEventTap,
    place: .headInsertEventTap,
    options: .defaultTap,
    eventsOfInterest: mask,
    callback: callback,
    userInfo: nil
) else {
    emit("no_permission")
    exit(1)
}

let source = CFMachPortCreateRunLoopSource(kCFAllocatorDefault, tap, 0)
CFRunLoopAddSource(CFRunLoopGetCurrent(), source, .commonModes)
CGEvent.tapEnable(tap: tap, enable: true)
emit("ready")
CFRunLoopRun()
```

`app/src/main/hotkeys.js`:

```js
import { spawn } from 'child_process'

const MIN_PTT_MS = 300

export function createHotkeyMachine({ onStart, onStop, onCancel, now = () => Date.now() }) {
  let state = 'idle' // idle | recording
  let source = null  // toggle | ptt
  let startedAt = 0

  function start(src) {
    state = 'recording'
    source = src
    startedAt = now()
    onStart()
  }

  function finish() {
    state = 'idle'
    source = null
    onStop()
  }

  function cancel() {
    state = 'idle'
    source = null
    onCancel()
  }

  return {
    state: () => state,
    handle(event) {
      if (event === 'toggle') {
        if (state === 'idle') start('toggle')
        else finish()
      } else if (event === 'ptt_down') {
        if (state === 'idle') start('ptt')
      } else if (event === 'ptt_up') {
        if (state === 'recording' && source === 'ptt') {
          if (now() - startedAt < MIN_PTT_MS) cancel()
          else finish()
        }
      } else if (event === 'ptt_abort') {
        if (state === 'recording' && source === 'ptt') cancel()
      } else if (event === 'esc') {
        if (state === 'recording') cancel()
      }
    }
  }
}

export function startHotkeys({ helperPath, machine, onNoPermission = () => {} }) {
  const proc = spawn(helperPath, [], { stdio: ['pipe', 'pipe', 'inherit'] })
  let buffer = ''
  proc.stdout.on('data', (chunk) => {
    buffer += chunk.toString()
    let idx
    while ((idx = buffer.indexOf('\n')) >= 0) {
      const line = buffer.slice(0, idx).trim()
      buffer = buffer.slice(idx + 1)
      if (!line) continue
      try {
        const { event } = JSON.parse(line)
        if (event === 'no_permission') onNoPermission()
        else if (event !== 'ready' && event !== 'combo') machine.handle(event)
      } catch { /* linha inválida, ignora */ }
    }
  })
  return {
    setRecordingState(on) { proc.stdin.write(`state ${on ? 'on' : 'off'}\n`) },
    kill() { proc.kill() }
  }
}
```

```jsx
import { useEffect, useRef, useState } from 'react'

export default function Popup() {
  const [phase, setPhase] = useState('idle') // idle|recording|processing|done|error
  const [result, setResult] = useState('')
  const canvasRef = useRef(null)
  const mediaRef = useRef({})

  useEffect(() => {
    window.dita.onCommand(async (ch, payload) => {
      if (ch === 'rec:start') await startRecording()
      if (ch === 'rec:stop') stopRecording(true)
      if (ch === 'rec:cancel') stopRecording(false)
      if (ch === 'rec:phase') setPhase(payload)
      if (ch === 'rec:result') { setResult(payload.text); setPhase('done') }
    })
  }, [])

  async function startRecording() {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
    const recorder = new MediaRecorder(stream, { mimeType: 'audio/webm' })
    const chunks = []
    recorder.ondataavailable = (e) => chunks.push(e.data)
    recorder.onstop = async () => {
      stream.getTracks().forEach((t) => t.stop())
      if (mediaRef.current.keep) {
        const buf = await new Blob(chunks).arrayBuffer()
        window.dita.sendAudio(buf)
      }
    }
    recorder.start(250)
    mediaRef.current = { recorder, stream, keep: false }
    setPhase('recording')
    drawWave(stream)
  }

  function stopRecording(keep) {
    const m = mediaRef.current
    if (m.recorder && m.recorder.state !== 'inactive') {
      m.keep = keep
      m.recorder.stop()
    }
    if (m.raf) cancelAnimationFrame(m.raf)
    setPhase(keep ? 'processing' : 'idle')
  }

  function drawWave(stream) {
    const ctx2d = canvasRef.current.getContext('2d')
    const audioCtx = new AudioContext()
    const analyser = audioCtx.createAnalyser()
    analyser.fftSize = 64
    audioCtx.createMediaStreamSource(stream).connect(analyser)
    const data = new Uint8Array(analyser.frequencyBinCount)
    const loop = () => {
      analyser.getByteFrequencyData(data)
      ctx2d.clearRect(0, 0, 180, 40)
      ctx2d.fillStyle = '#f5f5f4'
      for (let i = 0; i < 24; i++) {
        const h = Math.max(3, (data[i] / 255) * 34)
        ctx2d.fillRect(i * 7.5, 20 - h / 2, 4, h)
      }
      mediaRef.current.raf = requestAnimationFrame(loop)
    }
    loop()
  }

  const pill = {
    height: 56, margin: 8, borderRadius: 28, background: '#141414',
    display: 'flex', alignItems: 'center', padding: '0 14px', gap: 10,
    color: '#eee', fontFamily: 'system-ui', fontSize: 13,
    boxShadow: '0 6px 24px rgba(0,0,0,.45)'
  }

  if (phase === 'done') {
    return (
      <div style={{ ...pill, justifyContent: 'space-between' }}>
        <span style={{ overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{result}</span>
        <button
          onClick={() => { window.dita.copyText(result); window.dita.closePopup() }}
          style={{ background: '#3a3a3a', color: '#fff', border: 0, borderRadius: 8, padding: '6px 12px', cursor: 'pointer' }}
        >Copiar Ditado</button>
      </div>
    )
  }

  return (
    <div style={pill}>
      <button
        onClick={() => window.dita.closePopup()}
        style={{ background: '#2a2a2a', color: '#bbb', border: 0, borderRadius: '50%', width: 26, height: 26, cursor: 'pointer' }}

      {phase === 'processing'
        ? <span style={{ flex: 1, textAlign: 'center', color: '#aaa' }}>Transcrevendo…</span>

  )
```

```js
import { execFileSync } from 'child_process'
import { homedir, tmpdir } from 'os'
import { join } from 'path'

const FFMPEG = '/opt/homebrew/bin/ffmpeg'
const WHISPER_SERVER = '/opt/homebrew/opt/whisper-cpp/bin/whisper-server'
const MODEL_DIR = join(homedir(), 'Library/Application Support/dita-ai/models')
export const MODEL_PATH = join(MODEL_DIR, 'ggml-large-v3-turbo-q5_0.bin')
const MODEL_URL = 'https://huggingface.co/ggerganov/whisper.cpp/resolve/main/ggml-large-v3-turbo-q5_0.bin'
const PORT = 8971

export function ensureModel() {
  if (!existsSync(MODEL_PATH)) {
    mkdirSync(MODEL_DIR, { recursive: true })
    execFileSync('curl', ['-L', '-o', MODEL_PATH + '.part', MODEL_URL], { stdio: 'inherit' })
    execFileSync('mv', [MODEL_PATH + '.part', MODEL_PATH])
  }
  return MODEL_PATH

export async function startWhisperServer({ modelPath = MODEL_PATH, language = 'pt', port = PORT } = {}) {
  const proc = spawn(WHISPER_SERVER, [
    '--model', modelPath, '--host', '127.0.0.1', '--port', String(port),
    '--language', language, '--threads', '4'
  ], { stdio: ['ignore', 'pipe', 'pipe'] })

  // espera o server subir (tenta por até 30s: modelo grande demora a carregar)
  const deadline = Date.now() + 30000
  while (Date.now() < deadline) {
    try {
      await fetch(`http://127.0.0.1:${port}/`, { method: 'GET' })
      return { port, kill: () => proc.kill() }
    } catch { await new Promise((r) => setTimeout(r, 500)) }
  }
  proc.kill()
  throw new Error('whisper-server não subiu em 30s')
}

async function webmToWav(webmBuffer) {
  const base = join(tmpdir(), `dita-${Date.now()}`)
  const webm = `${base}.webm`
  const wav = `${base}.wav`
  await writeFile(webm, Buffer.from(webmBuffer))
  await new Promise((resolve, reject) => {
    execFile(FFMPEG, ['-y', '-i', webm, '-ar', '16000', '-ac', '1', '-c:a', 'pcm_s16le', wav],
      (err) => (err ? reject(err) : resolve()))
  })
  await unlink(webm).catch(() => {})
  return wav
}

export async function transcribe(webmBuffer, { port = PORT } = {}) {
  const wav = await webmToWav(webmBuffer)
  try {
    const form = new FormData()
    form.append('file', new Blob([await readFile(wav)]), 'audio.wav')
    form.append('response_format', 'json')
    const res = await fetch(`http://127.0.0.1:${port}/inference`, { method: 'POST', body: form })
    if (!res.ok) throw new Error(`whisper HTTP ${res.status}`)
    const data = await res.json()
    return (data.text || '').trim()
  } finally {
    await unlink(wav).catch(() => {})
  }
}
```

```js
function osascript(script) {

  })

  } catch {
    return false // sem permissão de acessibilidade ou keystroke falhou
  }
  // restaura o clipboard do usuário depois que o paste assentou
  setTimeout(() => {
    if (clipboard.readText() === text && previous) clipboard.writeText(previous)
  }, 400)
  return true
}
```

```js
export function createTray({ getLast, onPasteLast, onLanguageChange = () => {}, getLanguage = () => 'pt' }) {
  const tray = new Tray(ICON.resize({ width: 16, height: 16 }))
  tray.setToolTip('Dita Aí')

  function truncate(t) { return t && t.length > 30 ? t.slice(0, 30) + '…' : (t || 'Nenhum ditado ainda') }

  function buildMenu(lastText) {
    return Menu.buildFromTemplate([
      { label: 'Paste last transcript', accelerator: 'Ctrl+Cmd+V', click: onPasteLast },
      { label: truncate(lastText), enabled: false },
      { type: 'separator' },
      {
        label: 'Idiomas',
        submenu: [
          { label: 'Português', type: 'radio', checked: getLanguage() === 'pt', click: () => onLanguageChange('pt') },
          { label: 'English', type: 'radio', checked: getLanguage() === 'en', click: () => onLanguageChange('en') },
          { label: 'Automático', type: 'radio', checked: getLanguage() === 'auto', click: () => onLanguageChange('auto') }
        ]
      },
      { type: 'separator' },
      { label: 'Sair do Dita Aí', accelerator: 'Cmd+Q', click: () => app.quit() }
    ])
  }

  tray.setContextMenu(buildMenu(getLast()?.text))
  globalShortcut.register('Control+Command+V', onPasteLast)

  return {
    updateLastTranscript(text) { tray.setContextMenu(buildMenu(text)) }
  }
}
```
